     1→'use client';
     2→
     3→import { useRef, useState, useCallback } from 'react';
     4→import { RepoPicker } from '@/components/RepoPicker';
     5→import { CreateRepoDialog } from '@/components/CreateRepoDialog';
     6→import { Dropzone } from '@/components/Dropzone';
     7→import { FileTreePreview } from '@/components/FileTreePreview';
     8→import { ProgressBar, type ProgressStage } from '@/components/ProgressBar';
     9→import { WipeRepoButton } from '@/components/WipeRepoButton';
    10→import { SuccessScreen } from '@/components/SuccessScreen';
    11→import { AnalysisSection } from '@/components/AnalysisSection';
    12→import FilePreview from '@/components/FilePreview';
    13→import { useAppStore, type PushMode } from '@/store';
    14→import { uint8ArrayToBase64 } from '@/lib/zip';
    15→import { Button } from '@/components/ui/button';
    16→import { Input } from '@/components/ui/input';
    17→import { Label } from '@/components/ui/label';
    18→import { Textarea } from '@/components/ui/textarea';
    19→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    20→import { Badge } from '@/components/ui/badge';
    21→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    22→import { Separator } from '@/components/ui/separator';
    23→import {
    24→  Select,
    25→  SelectContent,
    26→  SelectItem,
    27→  SelectTrigger,
    28→  SelectValue,
    29→} from '@/components/ui/select';
    30→import {
    31→  Eye,
    32→  EyeOff,
    33→  KeyRound,
    34→  ArrowRight,
    35→  Upload,
    36→  GitBranch,
    37→  Shield,
    38→  Zap,
    39→  Github,
    40→  Plus,
    41→  Loader2,
    42→  Wand2,
    43→} from 'lucide-react';
    44→import { toast } from 'sonner';
    45→import type { GitHubRepo } from '@/lib/github';
    46→import { motion, AnimatePresence } from 'framer-motion';
    47→
    48→const fadeUp = {
    49→  initial: { opacity: 0, y: 12 },
    50→  animate: { opacity: 1, y: 0 },
    51→  exit: { opacity: 0, y: -8 },
    52→  transition: { duration: 0.3, ease: 'easeOut' },
    53→};
    54→
    55→export default function Home() {
    56→  const store = useAppStore();
    57→  const [showToken, setShowToken] = useState(false);
    58→  const [createRepoOpen, setCreateRepoOpen] = useState(false);
    59→  const [loadingBranches, setLoadingBranches] = useState(false);
    60→  const [newBranch, setNewBranch] = useState('');
    61→  const abortRef = useRef<AbortController | null>(null);
    62→
    63→  // Auto-generate commit message
    64→  if (store.files.length > 0 && !store.commitMessage) {
    65→    const date = new Date().toISOString().split('T')[0];
    66→    store.setCommitMessage(`Upload project - ${date}`);
    67→  }
    68→
    69→  const canShowRepos = store.token.length > 0;
    70→  const canShowUpload = store.selectedRepo !== null;
    71→  const canShowConfigure = store.files.length > 0;
    72→  const selectedFilesForPush = store.files.filter((f) =>
    73→    store.selectedFilePaths.has(f.path),
    74→  );
    75→
    76→  // Fetch branches when repo is selected
    77→  const fetchBranches = useCallback(async () => {
    78→    if (!store.selectedRepo || !store.token) return;
    79→    setLoadingBranches(true);
    80→    try {
    81→      const [owner, repo] = store.selectedRepo.full_name.split('/');
    82→      const res = await fetch(
    83→        `/api/branches?owner=${owner}&repo=${repo}`,
    84→        { headers: { Authorization: `Bearer ${store.token}` } },
    85→      );
    86→      if (res.ok) {
    87→        const data = await res.json();
    88→        const branches = (data.branches || []).map((b: { name: string }) => b.name);
    89→        store.setBranches(branches);
    90→        if (branches.length > 0 && !store.branch) {
    91→          store.setBranch(branches[0]);
    92→        }
    93→      }
    94→    } catch {
    95→      // Silent fail — branches are optional
    96→    } finally {
    97→      setLoadingBranches(false);
    98→    }
    99→  }, [store.selectedRepo, store.token, store.branch, store.setBranches, store.setBranch]);
   100→
   101→  // Watch for repo selection to load branches
   102→  const prevRepoRef = useRef<string>('');
   103→  if (store.selectedRepo?.full_name !== prevRepoRef.current) {
   104→    prevRepoRef.current = store.selectedRepo?.full_name || '';
   105→    if (store.selectedRepo) fetchBranches();
   106→  }
   107→
   108→  const handlePush = useCallback(async () => {
   109→    if (!store.selectedRepo || selectedFilesForPush.length === 0) return;
   110→
   111→    const [owner, repo] = store.selectedRepo.full_name.split('/');
   112→    if (!owner || !repo) { toast.error('Invalid repository selected'); return; }
   113→
   114→    store.setStage('pushing');
   115→    store.setPushError('');
   116→
   117→    const filesPayload = selectedFilesForPush.map((f) => ({
   118→      path: f.path,
   119→      content: uint8ArrayToBase64(f.content),
   120→      size: f.size,
   121→    }));
   122→
   123→    abortRef.current = new AbortController();
   124→
   125→    try {
   126→      const res = await fetch('/api/push', {
   127→        method: 'POST',
   128→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${store.token}` },
   129→        body: JSON.stringify({
   130→          owner, repo,
   131→          files: filesPayload,
   132→          commitMessage: store.commitMessage || 'Upload project',
   133→          mode: store.mode,
   134→          destination: store.destination.trim(),
   135→          branch: store.branch || undefined,
   136→        }),
   137→        signal: abortRef.current.signal,
   138→      });
   139→
   140→      if (!res.ok) {
   141→        const data = await res.json().catch(() => ({ error: 'Push failed' }));
   142→        throw new Error(data.error || `Push failed (${res.status})`);
   143→      }
   144→
   145→      const reader = res.body?.getReader();
   146→      if (!reader) throw new Error('No response body');
   147→
   148→      const decoder = new TextDecoder();
   149→      let buffer = '';
   150→      const stageMap = new Map<string, number>();
   151→      let stageList: ProgressStage[] = [];
   152→      let currentIdx = 0;
   153→
   154→      while (true) {
   155→        const { done, value } = await reader.read();
   156→        if (done) break;
   157→        buffer += decoder.decode(value, { stream: true });
   158→        const lines = buffer.split('\n');
   159→        buffer = lines.pop() || '';
   160→
   161→        for (const line of lines) {
   162→          if (!line.startsWith('data: ')) continue;
   163→          try {
   164→            const event = JSON.parse(line.slice(6));
   165→            if (event.type === 'progress') {
   166→              if (!stageMap.has(event.stage)) {
   167→                stageMap.set(event.stage, stageList.length);
   168→                stageList.push({ stage: event.stage, message: event.message, current: event.current, total: event.total });
   169→              } else {
   170→                const idx = stageMap.get(event.stage)!;
   171→                stageList[idx] = { stage: event.stage, message: event.message, current: event.current, total: event.total };
   172→              }
   173→              currentIdx = stageMap.get(event.stage) || 0;
   174→              store.setProgress(stageList, currentIdx);
   175→            } else if (event.type === 'success') {
   176→              store.setProgress(stageList, stageList.length);
   177→              store.setSuccessData({
   178→                commitSha: event.commitSha, commitUrl: event.commitUrl,
   179→                repoUrl: event.repoUrl, filesUploaded: event.filesUploaded,
   180→                filesChanged: event.filesChanged, commitMessage: store.commitMessage,
   181→              });
   182→              store.setStage('success');
   183→              toast.success('Successfully pushed to GitHub!');
   184→            } else if (event.type === 'error') {
   185→              store.setPushError(event.error);
   186→              toast.error(event.error);
   187→            }
   188→          } catch { /* ignore */ }
   189→        }
   190→      }
   191→    } catch (e) {
   192→      if ((e as Error).name === 'AbortError') return;
   193→      const msg = e instanceof Error ? e.message : 'An unexpected error occurred';
   194→      store.setPushError(msg);
   195→      toast.error(msg);
   196→    }
   197→  }, [store, selectedFilesForPush]);
   198→
   199→  const handleFilesReady = useCallback(
   200→    (files: typeof store.files, totalSize: number, errors: string[]) => {
   201→      store.setFiles(files, totalSize, errors);
   202→      if (files.length > 0) store.setStage('configure');
   203→    },
   204→    [store],
   205→  );
   206→
   207→  const handleRepoCreated = useCallback(
   208→    (repo: GitHubRepo) => {
   209→      store.setRepos([repo, ...store.repos]);
   210→      store.setSelectedRepo(repo);
   211→      toast.success(`Repository ${repo.name} created!`);
   212→    },
   213→    [store],
   214→  );
   215→
   216→  const handleCreateBranch = useCallback(async () => {
   217→    if (!newBranch.trim()) return;
   218→    const name = newBranch.trim();
   219→    store.setBranches([...store.branches, name]);
   220→    store.setBranch(name);
   221→    setNewBranch('');
   222→    toast.success(`Will push to new branch: ${name}`);
   223→  }, [newBranch, store]);
   224→
   225→  return (
   226→    <div className="mx-auto max-w-3xl px-4 py-8 space-y-6">
   227→      {/* Hero */}
   228→      <motion.div
   229→        className="text-center space-y-3"
   230→        initial={{ opacity: 0, y: -10 }}
   231→        animate={{ opacity: 1, y: 0 }}
   232→        transition={{ duration: 0.4 }}
   233→      >
   234→        <h1 className="text-3xl sm:text-4xl font-bold tracking-tight">
   235→          Drop your project.{' '}
   236→          <span className="text-primary">Push</span> it to GitHub.
   237→          <span className="text-primary"> Done.</span>
   238→        </h1>
   239→        <p className="text-muted-foreground max-w-md mx-auto">
   240→          Upload projects directly to GitHub without the terminal.
   241→          No Git commands. No CLI. Just drag, drop, and push.
   242→        </p>
   243→      </motion.div>
   244→
   245→      <div className="space-y-6">
   246→        {/* Step 1: Credentials */}
   247→        <motion.section {...fadeUp} layout>
   248→          <Card className="overflow-hidden">
   249→            <CardHeader className="pb-4">
   250→              <div className="flex items-center gap-3">
   251→                <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10">
   252→                  <KeyRound className="h-4 w-4 text-primary" />
   253→                </div>
   254→                <div>
   255→                  <CardTitle className="text-base">GitHub Personal Access Token</CardTitle>
   256→                  <CardDescription>Required to authenticate with the GitHub API.</CardDescription>
   257→                </div>
   258→              </div>
   259→            </CardHeader>
   260→            <CardContent className="space-y-3">
   261→              <div className="relative">
   262→                <Input
   263→                  type={showToken ? 'text' : 'password'}
   264→                  placeholder="ghp_xxxxxxxxxxxxxxxxxxxx"
   265→                  value={store.token}
   266→                  onChange={(e) => store.setToken(e.target.value.trim())}
   267→                  className="pr-10 font-mono text-sm"
   268→                  aria-label="GitHub Personal Access Token"
   269→                />
   270→                <button
   271→                  type="button"
   272→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition-colors"
   273→                  onClick={() => setShowToken(!showToken)}
   274→                  aria-label={showToken ? 'Hide token' : 'Show token'}
   275→                >
   276→                  {showToken ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   277→                </button>
   278→              </div>
   279→              <div className="flex items-start gap-2 text-xs text-muted-foreground bg-muted/50 rounded-lg p-3">
   280→                <Shield className="h-3.5 w-3.5 mt-0.5 shrink-0 text-primary" />
   281→                <p>
   282→                  Your token is used <strong>only for this session</strong> and is never stored,
   283→                  logged, or shared. It is sent directly to the GitHub API over HTTPS.
   284→                </p>
   285→              </div>
   286→            </CardContent>
   287→          </Card>
   288→        </motion.section>
   289→
   290→        {/* Step 2: Repository */}
   291→        <AnimatePresence>
   292→          {canShowRepos && (
   293→            <motion.section {...fadeUp} layout>
   294→              <Card className="overflow-hidden">
   295→                <CardHeader className="pb-4">
   296→                  <div className="flex items-center gap-3">
   297→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-sky-accent/10">
   298→                      <GitBranch className="h-4 w-4 text-sky-accent" />
   299→                    </div>
   300→                    <div>
   301→                      <CardTitle className="text-base">Repository</CardTitle>
   302→                      <CardDescription>Select or create a repository to push to.</CardDescription>
   303→                    </div>
   304→                  </div>
   305→                </CardHeader>
   306→                <CardContent className="space-y-3">
   307→                  <RepoPicker
   308→                    token={store.token}
   309→                    selectedRepo={store.selectedRepo}
   310→                    onSelect={store.setSelectedRepo}
   311→                    onCreateRepo={() => setCreateRepoOpen(true)}
   312→                    disabled={store.stage === 'pushing'}
   313→                  />
   314→                  {store.selectedRepo && (
   315→                    <div className="flex items-center justify-between flex-wrap gap-2">
   316→                      <p className="text-xs text-muted-foreground">
   317→                        <Github className="inline h-3 w-3 mr-1" />
   318→                        <a href={store.selectedRepo.html_url} target="_blank" rel="noopener noreferrer" className="text-primary hover:underline">
   319→                          {store.selectedRepo.full_name}
   320→                        </a>
   321→                      </p>
   322→                      <WipeRepoButton
   323→                        owner={store.selectedRepo.full_name.split('/')[0]}
   324→                        repoName={store.selectedRepo.name}
   325→                        token={store.token}
   326→                        disabled={store.stage === 'pushing'}
   327→                        onWiped={() => toast.success('All files deleted successfully')}
   328→                      />
   329→                    </div>
   330→                  )}
   331→                </CardContent>
   332→              </Card>
   333→            </motion.section>
   334→          )}
   335→        </AnimatePresence>
   336→
   337→        {/* Step 3: Upload */}
   338→        <AnimatePresence>
   339→          {canShowUpload && store.stage !== 'success' && (
   340→            <motion.section {...fadeUp} layout>
   341→              <Card className="overflow-hidden">
   342→                <CardHeader className="pb-4">
   343→                  <div className="flex items-center gap-3">
   344→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10">
   345→                      <Upload className="h-4 w-4 text-primary" />
   346→                    </div>
   347→                    <div>
   348→                      <CardTitle className="text-base">Upload Project</CardTitle>
   349→                      <CardDescription>Drag and drop a .zip file or folder.</CardDescription>
   350→                    </div>
   351→                  </div>
   352→                </CardHeader>
   353→                <CardContent><Dropzone onFilesReady={handleFilesReady} isProcessing={store.stage === 'pushing'} /></CardContent>
   354→              </Card>
   355→            </motion.section>
   356→          )}
   357→        </AnimatePresence>
   358→
   359→        {/* Step 3.5: Analysis */}
   360→        <AnimatePresence>
   361→          {canShowConfigure && store.stage !== 'success' && (
   362→            <motion.section {...fadeUp} layout>
   363→              <AnalysisSection />
   364→            </motion.section>
   365→          )}
   366→        </AnimatePresence>
   367→
   368→        {/* Step 4: Configure & Push */}
   369→        <AnimatePresence>
   370→          {canShowConfigure && store.stage !== 'success' && (
   371→            <motion.section {...fadeUp} layout>
   372→              <Card className="overflow-hidden">
   373→                <CardHeader className="pb-4">
   374→                  <div className="flex items-center gap-3">
   375→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10">
   376→                      <Zap className="h-4 w-4 text-primary" />
   377→                    </div>
   378→                    <div>
   379→                      <CardTitle className="text-base">Configure & Push</CardTitle>
   380→                      <CardDescription>Review your files and push to GitHub.</CardDescription>
   381→                    </div>
   382→                  </div>
   383→                </CardHeader>
   384→                <CardContent className="space-y-5">
   385→                  {/* File Tree with checkboxes */}
   386→                  <div>
   387→                    <label className="text-sm font-medium mb-2 block">Files Preview <span className="text-muted-foreground font-normal">(click a file to preview)</span></label>
   388→                    <FileTreePreview
   389→                      files={store.files}
   390→                      selectable
   391→                      selectedPaths={store.selectedFilePaths}
   392→                      onToggleFile={store.toggleFilePath}
   393→                      onFileClick={store.setPreviewFile}
   394→                    />
   395→                  </div>
   396→
   397→                  <Separator />
   398→
   399→                  {/* Branch Selector */}
   400→                  <div className="space-y-2">
   401→                    <Label>Branch</Label>
   402→                    <div className="flex gap-2">
   403→                      <Select
   404→                        value={store.branch}
   405→                        onValueChange={store.setBranch}
   406→                      >
   407→                        <SelectTrigger className="flex-1 font-mono text-sm">
   408→                          <SelectValue placeholder={loadingBranches ? 'Loading…' : 'Select branch'} />
   409→                        </SelectTrigger>
   410→                        <SelectContent>
   411→                          {store.branches.map((b) => (
   412→                            <SelectItem key={b} value={b} className="font-mono">{b}</SelectItem>
   413→                          ))}
   414→                        </SelectContent>
   415→                      </Select>
   416→                      <div className="relative flex-1">
   417→                        <Input
   418→                          placeholder="new-branch-name"
   419→                          value={newBranch}
   420→                          onChange={(e) => setNewBranch(e.target.value)}
   421→                          className="font-mono text-sm pr-9"
   422→                          onKeyDown={(e) => e.key === 'Enter' && handleCreateBranch()}
   423→                        />
   424→                        <button
   425→                          type="button"
   426→                          className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-primary transition-colors"
   427→                          onClick={handleCreateBranch}
   428→                          disabled={!newBranch.trim()}
   429→                          aria-label="Create new branch"
   430→                        >
   431→                          <Plus className="h-4 w-4" />
   432→                        </button>
   433→                      </div>
   434→                    </div>
   435→                  </div>
   436→
   437→                  {/* Push Mode */}
   438→                  <div className="space-y-3">
   439→                    <Label>Push Mode</Label>
   440→                    <RadioGroup
   441→                      value={store.mode}
   442→                      onValueChange={(v) => store.setMode(v as PushMode)}
   443→                      className="grid grid-cols-1 sm:grid-cols-2 gap-3"
   444→                    >
   445→                      <label
   446→                        htmlFor="mode-replace"
   447→                        className={`relative flex cursor-pointer flex-col rounded-lg border-2 p-4 transition-all hover:bg-muted/30 ${
   448→                          store.mode === 'replace' ? 'border-primary bg-primary/5' : 'border-muted'
   449→                        }`}
   450→                      >
   451→                        <div className="flex items-start gap-2.5">
   452→                          <RadioGroupItem value="replace" id="mode-replace" className="mt-0.5" />
   453→                          <div className="space-y-1">
   454→                            <p className="text-sm font-semibold">Replace Everything</p>
   455→                            <p className="text-xs text-muted-foreground">
   456→                              Completely replace the repository contents. Old files will be removed in a single clean commit.
   457→                            </p>
   458→                          </div>
   459→                        </div>
   460→                        {store.mode === 'replace' && (
   461→                          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute inset-x-0 bottom-0 h-0.5 bg-primary rounded-b-lg" />
   462→                        )}
   463→                      </label>
   464→                      <label
   465→                        htmlFor="mode-smart"
   466→                        className={`relative flex cursor-pointer flex-col rounded-lg border-2 p-4 transition-all hover:bg-muted/30 ${
   467→                          store.mode === 'smart' ? 'border-sky-accent bg-sky-accent/5' : 'border-muted'
   468→                        }`}
   469→                      >
   470→                        <div className="flex items-start gap-2.5">
   471→                          <RadioGroupItem value="smart" id="mode-smart" className="mt-0.5" />
   472→                          <div className="space-y-1">
   473→                            <p className="text-sm font-semibold">Smart Update</p>
   474→                            <p className="text-xs text-muted-foreground">Only upload new and changed files. Existing files are preserved.</p>
   475→                          </div>
   476→                        </div>
   477→                        {store.mode === 'smart' && (
   478→                          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute inset-x-0 bottom-0 h-0.5 bg-sky-accent rounded-b-lg" />
   479→                        )}
   480→                      </label>
   481→                    </RadioGroup>
   482→                  </div>
   483→
   484→                  {store.mode === 'replace' && (
   485→                    <div className="flex items-start gap-2 text-xs text-amber-600 dark:text-amber-400 bg-amber-500/10 rounded-lg p-3">
   486→                      <Shield className="h-3.5 w-3.5 mt-0.5 shrink-0" />
   487→                      <p><strong>Destructive action:</strong> This will remove all existing files in the repository that are not in your upload. Make sure you have a backup if needed.</p>
   488→                    </div>
   489→                  )}
   490→
   491→                  {/* Destination Path */}
   492→                  <div className="space-y-2">
   493→                    <Label htmlFor="destination">Destination Subfolder (optional)</Label>
   494→                    <Input
   495→                      id="destination"
   496→                      placeholder="e.g., src/project — leave empty for root"
   497→                      value={store.destination}
   498→                      onChange={(e) => store.setDestination(e.target.value)}
   499→                      className="font-mono text-sm"
   500→                    />
   501→                  </div>
   502→
   503→                  {/* Commit Message */}
   504→                  <div className="space-y-2">
   505→                    <div className="flex items-center justify-between">
   506→                      <Label htmlFor="commit-msg">Commit Message</Label>
   507→                      <Button
   508→                        variant="ghost"
   509→                        size="sm"
   510→                        className="h-6 text-xs gap-1 px-2"
   511→                        onClick={() => {
   512→                          const date = new Date().toISOString().split('T')[0];
   513→                          store.setCommitMessage(`Upload project - ${date}`);
   514→                          toast.success('Suggested a commit message');
   515→                        }}
   516→                      >
   517→                        <Wand2 className="h-3 w-3" />
   518→                        Suggest
   519→                      </Button>
   520→                    </div>
   521→                    <Textarea
   522→                      id="commit-msg"
   523→                      placeholder="Describe your changes..."
   524→                      value={store.commitMessage}
   525→                      onChange={(e) => store.setCommitMessage(e.target.value)}
   526→                      className="min-h-[80px] resize-y text-sm"
   527→                    />
   528→                  </div>
   529→
   530→                  {/* Stats */}
   531→                  <div className="flex flex-wrap gap-2">
   532→                    <Badge variant="secondary" className="text-xs">
   533→                      {selectedFilesForPush.length}/{store.files.length} files selected
   534→                    </Badge>
   535→                    <Badge variant="secondary" className="text-xs">{store.mode === 'replace' ? 'Replace' : 'Smart Update'}</Badge>
   536→                    {store.branch && (
   537→                      <Badge variant="secondary" className="text-xs font-mono">
   538→                        <GitBranch className="mr-1 h-3 w-3" />{store.branch}
   539→                      </Badge>
   540→                    )}
   541→                    {store.destination && (
   542→                      <Badge variant="secondary" className="text-xs font-mono">→ {store.destination}</Badge>
   543→                    )}
   544→                  </div>
   545→
   546→                  {/* Push Button */}
   547→                  <Button
   548→                    size="lg"
   549→                    className="w-full h-12 text-base font-semibold"
   550→                    onClick={handlePush}
   551→                    disabled={
   552→                      !store.selectedRepo ||
   553→                      selectedFilesForPush.length === 0 ||
   554→                      !store.commitMessage.trim() ||
   555→                      store.stage === 'pushing'
   556→                    }
   557→                  >
   558→                    <Upload className="mr-2 h-5 w-5" />
   559→                    Push {selectedFilesForPush.length} File{selectedFilesForPush.length !== 1 ? 's' : ''} to GitHub
   560→                    <ArrowRight className="ml-2 h-4 w-4" />
   561→                  </Button>
   562→                </CardContent>
   563→              </Card>
   564→            </motion.section>
   565→          )}
   566→        </AnimatePresence>
   567→
   568→        {/* Step 5: Progress */}
   569→        <AnimatePresence>
   570→          {store.stage === 'pushing' && (
   571→            <motion.section {...fadeUp} layout>
   572→              <ProgressBar
   573→                stages={store.progressStages}
   574→                currentStage={store.currentProgressStage}
   575→                isComplete={false}
   576→                error={store.pushError}
   577→              />
   578→              {store.pushError && (
   579→                <Button variant="outline" className="w-full mt-4" onClick={() => store.setStage('configure')}>
   580→                  Go Back and Retry
   581→                </Button>
   582→              )}
   583→            </motion.section>
   584→          )}
   585→        </AnimatePresence>
   586→
   587→        {/* Step 6: Success */}
   588→        <AnimatePresence>
   589→          {store.stage === 'success' && store.successData && (
   590→            <motion.section {...fadeUp} layout>
   591→              <SuccessScreen
   592→                repoUrl={store.successData.repoUrl}
   593→                commitSha={store.successData.commitSha}
   594→                commitUrl={store.successData.commitUrl}
   595→                commitMessage={store.successData.commitMessage}
   596→                filesUploaded={store.successData.filesUploaded}
   597→                filesChanged={store.successData.filesChanged}
   598→                onPushAnother={() => store.resetPush()}
   599→              />
   600→            </motion.section>
   601→          )}
   602→        </AnimatePresence>
   603→      </div>
   604→
   605→      {/* File Preview Dialog */}
   606→      <FilePreview
   607→        file={store.previewFile}
   608→        open={!!store.previewFile}
   609→        onOpenChange={(open) => { if (!open) store.setPreviewFile(null); }}
   610→      />
   611→
   612→      {/* Create Repo Dialog */}
   613→      <CreateRepoDialog
   614→        open={createRepoOpen}
   615→        onOpenChange={setCreateRepoOpen}
   616→        token={store.token}
   617→        onCreated={handleRepoCreated}
   618→      />
   619→    </div>
   620→  );
   621→}
