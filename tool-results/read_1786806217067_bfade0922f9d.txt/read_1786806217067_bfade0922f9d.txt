    62→  return (
    63→    <div className={cn(
    64→      'flex h-10 w-10 items-center justify-center rounded-xl shadow-sm',

    68→    )}>
    69→      {children}
    70→    </div>
    71→  );
    72→}
    73→
    74→export default function Home() {
    75→  const store = useAppStore();
    76→  const [showToken, setShowToken] = useState(false);
    77→  const [createRepoOpen, setCreateRepoOpen] = useState(false);
    78→  const [loadingBranches, setLoadingBranches] = useState(false);
    79→  const [newBranch, setNewBranch] = useState('');
    80→  const [uploadTab, setUploadTab] = useState<'local' | 'zai'>('local');
    81→  const abortRef = useRef<AbortController | null>(null);
    82→
    83→  // Auto-generate commit message
    84→  if (store.files.length > 0 && !store.commitMessage) {
    85→    const date = new Date().toISOString().split('T')[0];
    86→    store.setCommitMessage(`Upload project - ${date}`);
    87→  }
    88→
    89→  const canShowRepos = store.token.length > 0;
    90→  const canShowUpload = store.selectedRepo !== null;
    91→  const canShowConfigure = store.files.length > 0;
    92→  const selectedFilesForPush = store.files.filter((f) =>
    93→    store.selectedFilePaths.has(f.path),
    94→  );
    95→
    96→  // Fetch branches when repo is selected
    97→  const fetchBranches = useCallback(async () => {
    98→    if (!store.selectedRepo || !store.token) return;
    99→    setLoadingBranches(true);
   100→    try {
   101→      const [owner, repo] = store.selectedRepo.full_name.split('/');
   102→      const res = await fetch(
   103→        `/api/branches?owner=${owner}&repo=${repo}`,
   104→        { headers: { Authorization: `Bearer ${store.token}` } },
   105→      );
   106→      if (res.ok) {
   107→        const data = await res.json();
   108→        const branches = (data.branches || []).map((b: { name: string }) => b.name);
   109→        store.setBranches(branches);
   110→        if (branches.length > 0 && !store.branch) {
   111→          store.setBranch(branches[0]);
   112→        }
   113→      }
   114→    } catch {
   115→      // Silent fail — branches are optional
   116→    } finally {
   117→      setLoadingBranches(false);
   118→    }
   119→  }, [store.selectedRepo, store.token, store.branch, store.setBranches, store.setBranch]);
   120→
   121→  // Watch for repo selection to load branches
   122→  const prevRepoRef = useRef<string>('');
   123→  if (store.selectedRepo?.full_name !== prevRepoRef.current) {
   124→    prevRepoRef.current = store.selectedRepo?.full_name || '';
   125→    if (store.selectedRepo) fetchBranches();
   126→  }
   127→
   128→  const handlePush = useCallback(async () => {
   129→    if (!store.selectedRepo || selectedFilesForPush.length === 0) return;
   130→
   131→    const [owner, repo] = store.selectedRepo.full_name.split('/');
   132→    if (!owner || !repo) { toast.error('Invalid repository selected'); return; }
   133→
   134→    store.setStage('pushing');
   135→    store.setPushError('');
   136→
   137→    const filesPayload = selectedFilesForPush.map((f) => ({
   138→      path: f.path,
   139→      content: uint8ArrayToBase64(f.content),
   140→      size: f.size,
   141→    }));
   142→
   143→    abortRef.current = new AbortController();
   144→
   145→    try {
   146→      const res = await fetch('/api/push', {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${store.token}` },
   149→        body: JSON.stringify({
   150→          owner, repo,
   151→          files: filesPayload,
   152→          commitMessage: store.commitMessage || 'Upload project',
   153→          mode: store.mode,
   154→          destination: store.destination.trim(),
   155→          branch: store.branch || undefined,
   156→        }),
   157→        signal: abortRef.current.signal,
   158→      });
   159→
   160→      if (!res.ok) {
   161→        const data = await res.json().catch(() => ({ error: 'Push failed' }));
   162→        throw new Error(data.error || `Push failed (${res.status})`);
   163→      }
   164→
   165→      const reader = res.body?.getReader();
   166→      if (!reader) throw new Error('No response body');
   167→
   168→      const decoder = new TextDecoder();
   169→      let buffer = '';
   170→      const stageMap = new Map<string, number>();
   171→      let stageList: ProgressStage[] = [];
   172→      let currentIdx = 0;
   173→
   174→      while (true) {
   175→        const { done, value } = await reader.read();
   176→        if (done) break;
   177→        buffer += decoder.decode(value, { stream: true });
   178→        const lines = buffer.split('\n');
   179→        buffer = lines.pop() || '';
   180→
   181→        for (const line of lines) {
   182→          if (!line.startsWith('data: ')) continue;
   183→          try {
   184→            const event = JSON.parse(line.slice(6));
   185→            if (event.type === 'progress') {
   186→              if (!stageMap.has(event.stage)) {
   187→                stageMap.set(event.stage, stageList.length);
   188→                stageList.push({ stage: event.stage, message: event.message, current: event.current, total: event.total });
   189→              } else {
   190→                const idx = stageMap.get(event.stage)!;
   191→                stageList[idx] = { stage: event.stage, message: event.message, current: event.current, total: event.total };
   192→              }
   193→              currentIdx = stageMap.get(event.stage) || 0;
   194→              store.setProgress(stageList, currentIdx);
   195→            } else if (event.type === 'success') {
   196→              store.setProgress(stageList, stageList.length);
   197→              store.setSuccessData({
   198→                commitSha: event.commitSha, commitUrl: event.commitUrl,
   199→                repoUrl: event.repoUrl, filesUploaded: event.filesUploaded,
   200→                filesChanged: event.filesChanged, commitMessage: store.commitMessage,
   201→              });
   202→              store.setStage('success');
   203→              toast.success('Successfully pushed to GitHub!');
   204→            } else if (event.type === 'error') {
   205→              store.setPushError(event.error);
   206→              toast.error(event.error);
   207→            }
   208→          } catch { /* ignore */ }
   209→        }
   210→      }
   211→    } catch (e) {
   212→      if ((e as Error).name === 'AbortError') return;
   213→      const msg = e instanceof Error ? e.message : 'An unexpected error occurred';
   214→      store.setPushError(msg);
   215→      toast.error(msg);
   216→    }
   217→  }, [store, selectedFilesForPush]);
   218→
   219→  const handleFilesReady = useCallback(
   220→    (files: typeof store.files, totalSize: number, errors: string[]) => {
   221→      store.setFiles(files, totalSize, errors);
   222→      if (files.length > 0) store.setStage('configure');
   223→    },
   224→    [store],
   225→  );
   226→
   227→  const handleRepoCreated = useCallback(
   228→    (repo: GitHubRepo) => {
   229→      store.setRepos([repo, ...store.repos]);
   230→      store.setSelectedRepo(repo);
   231→      toast.success(`Repository ${repo.name} created!`);
   232→    },
   233→    [store],
   234→  );
   235→
   236→  const handleCreateBranch = useCallback(async () => {
   237→    if (!newBranch.trim()) return;
   238→    const name = newBranch.trim();
   239→    store.setBranches([...store.branches, name]);
   240→    store.setBranch(name);
   241→    setNewBranch('');
   242→    toast.success(`Will push to new branch: ${name}`);
   243→  }, [newBranch, store]);
   244→
   245→  return (
   246→    <div className="mx-auto max-w-3xl px-4 py-8 space-y-6">
   247→      {/* ─── Hero ──────────────────────────────────────────── */}
   248→      <motion.div
   249→        className="text-center space-y-3"
   250→        initial={{ opacity: 0, y: -10 }}
   251→        animate={{ opacity: 1, y: 0 }}
   252→        transition={{ duration: 0.4 }}
   253→      >
   254→        <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-primary/10 border border-primary/20 text-xs font-medium text-primary mb-4">
   255→          <Zap className="h-3.5 w-3.5" />
   256→          Free &amp; Open Source
   257→        </div>
   258→        <h1 className="text-3xl sm:text-4xl font-bold tracking-tight">
   259→          Drop your project.{' '}
   260→          <span className="bg-gradient-to-r from-primary to-sky-accent bg-clip-text text-transparent">Push</span>{' '}
   261→          it to GitHub.
   262→        </h1>
   263→        <p className="text-muted-foreground max-w-md mx-auto text-sm leading-relaxed">
   264→          Upload projects directly to GitHub without the terminal.
   265→          No Git commands. No CLI. Just drag, drop, and push.
   266→        </p>
   267→      </motion.div>
   268→
   269→      <div className="space-y-6">
   270→        {/* ─── Step 1: GitHub Token ──────────────────────── */}
   271→        <motion.section {...fadeUp} layout>
   272→          <Card className="overflow-hidden ring-1 ring-border/50">
   273→            <CardHeader className="pb-4">
   274→              <div className="flex items-center gap-3">
   275→                <StepIcon><KeyRound className="h-5 w-5 text-primary" /></StepIcon>
   276→                <div>
   277→                  <CardTitle className="text-base">GitHub Personal Access Token</CardTitle>
   278→                  <CardDescription>Required to authenticate with the GitHub API.</CardDescription>
   279→                </div>
   280→              </div>
   281→            </CardHeader>
   282→            <CardContent className="space-y-3">
   283→              <div className="relative">
   284→                <Input
   285→                  type={showToken ? 'text' : 'password'}
   286→                  placeholder="ghp_xxxxxxxxxxxxxxxxxxxx"
   287→                  value={store.token}
   288→                  onChange={(e) => store.setToken(e.target.value.trim())}
   289→                  className="pr-10 font-mono text-sm"
   290→                  aria-label="GitHub Personal Access Token"
   291→                />
   292→                <button
   293→                  type="button"
   294→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition-colors"
   295→                  onClick={() => setShowToken(!showToken)}
   296→                  aria-label={showToken ? 'Hide token' : 'Show token'}
   297→                >
   298→                  {showToken ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   299→                </button>
   300→              </div>
   301→              <div className="flex items-start gap-2 text-xs text-muted-foreground bg-muted/50 rounded-lg p-3">
   302→                <Shield className="h-3.5 w-3.5 mt-0.5 shrink-0 text-primary" />
   303→                <p>
   304→                  Your token is used <strong>only for this session</strong> and is never stored,
   305→                  logged, or shared. It is sent directly to the GitHub API over HTTPS.
   306→                </p>
   307→              </div>
   308→            </CardContent>
   309→          </Card>
   310→        </motion.section>
   311→
   312→        {/* ─── Step 2: Repository ──────────────────────── */}
   313→        <AnimatePresence>
   314→          {canShowRepos && (
   315→            <motion.section {...fadeUp} layout>
   316→              <Card className="overflow-hidden ring-1 ring-border/50">
   317→                <CardHeader className="pb-4">
   318→                  <div className="flex items-center gap-3">
   319→                    <StepIcon color="accent"><GitBranch className="h-5 w-5 text-sky-accent" /></StepIcon>
   320→                    <div>
   321→                      <CardTitle className="text-base">Repository</CardTitle>
   322→                      <CardDescription>Select or create a repository to push to.</CardDescription>
   323→                    </div>
   324→                  </div>
   325→                </CardHeader>
   326→                <CardContent className="space-y-3">
   327→                  <RepoPicker
   328→                    token={store.token}
   329→                    selectedRepo={store.selectedRepo}
   330→                    onSelect={store.setSelectedRepo}
   331→                    onCreateRepo={() => setCreateRepoOpen(true)}
   332→                    disabled={store.stage === 'pushing'}
   333→                  />
   334→                  {store.selectedRepo && (
   335→                    <div className="flex items-center justify-between flex-wrap gap-2">
   336→                      <p className="text-xs text-muted-foreground">
   337→                        <Github className="inline h-3 w-3 mr-1" />
   338→                        <a href={store.selectedRepo.html_url} target="_blank" rel="noopener noreferrer" className="text-primary hover:underline">
   339→                          {store.selectedRepo.full_name}
   340→                        </a>
   341→                      </p>
   342→                      <WipeRepoButton
   343→                        owner={store.selectedRepo.full_name.split('/')[0]}
   344→                        repoName={store.selectedRepo.name}
   345→                        token={store.token}
   346→                        disabled={store.stage === 'pushing'}
   347→                        onWiped={() => toast.success('All files deleted successfully')}
   348→                      />
   349→                    </div>
   350→                  )}
   351→                </CardContent>
   352→              </Card>
   353→            </motion.section>
   354→          )}
   355→        </AnimatePresence>
   356→
   357→        {/* ─── Step 3: Upload (Tabs: Local / z.ai) ────── */}
   358→        <AnimatePresence>
   359→          {canShowUpload && store.stage !== 'success' && (
   360→            <motion.section {...fadeUp} layout>
   361→              <Card className="overflow-hidden ring-1 ring-border/50">
   362→                <CardHeader className="pb-4">
   363→                  <div className="flex items-center gap-3">
   364→                    <StepIcon><Upload className="h-5 w-5 text-primary" /></StepIcon>
   365→                    <div className="flex-1">
   366→                      <CardTitle className="text-base">Upload Project</CardTitle>
   367→                      <CardDescription>Drag &amp; drop files or load from z.ai.</CardDescription>
   368→                    </div>
   369→                  </div>
   370→
   371→                  {/* Source tabs */}
   372→                  <div className="flex gap-1 mt-3 p-1 bg-muted/60 rounded-lg w-fit">
   373→                    <button
   374→                      type="button"
   375→                      onClick={() => setUploadTab('local')}
   376→                      className={cn(
   377→                        'flex items-center gap-1.5 px-3 py-1.5 text-sm font-medium rounded-md transition-all',
   378→                        uploadTab === 'local'
   379→                          ? 'bg-background shadow-sm text-foreground'
   380→                          : 'text-muted-foreground hover:text-foreground',
   381→                      )}
   382→                    >
   383→                      <CloudUpload className="h-3.5 w-3.5" />
   384→                      Local Files
   385→                    </button>
   386→                    <button
   387→                      type="button"
   388→                      onClick={() => setUploadTab('zai')}
   389→                      className={cn(
   390→                        'flex items-center gap-1.5 px-3 py-1.5 text-sm font-medium rounded-md transition-all',
   391→                        uploadTab === 'zai'
   392→                          ? 'bg-background shadow-sm text-foreground'
   393→                          : 'text-muted-foreground hover:text-foreground',
   394→                      )}
   395→                    >
   396→                      <Cloud className="h-3.5 w-3.5" />
   397→                      z.ai
   398→                    </button>
   399→                  </div>
   400→                </CardHeader>
   401→                <CardContent>
   402→                  {uploadTab === 'local' ? (
   403→                    <Dropzone onFilesReady={handleFilesReady} isProcessing={store.stage === 'pushing'} />
   404→                  ) : (
   405→                    <ZaiLoader disabled={store.stage === 'pushing'} />
   406→                  )}
   407→                </CardContent>
   408→              </Card>
   409→            </motion.section>
   410→          )}
   411→        </AnimatePresence>
   412→
   413→        {/* ─── Step 3.5: Analysis ──────────────────────── */}
   414→        <AnimatePresence>
   415→          {canShowConfigure && store.stage !== 'success' && (
   416→            <motion.section {...fadeUp} layout>
   417→              <AnalysisSection />
   418→            </motion.section>
   419→          )}
   420→        </AnimatePresence>
   421→
   422→        {/* ─── Step 4: Configure & Push ─────────────────── */}
   423→        <AnimatePresence>
   424→          {canShowConfigure && store.stage !== 'success' && (
   425→            <motion.section {...fadeUp} layout>
   426→              <Card className="overflow-hidden ring-1 ring-border/50">
   427→                <CardHeader className="pb-4">
   428→                  <div className="flex items-center gap-3">
   429→                    <StepIcon color="accent"><Zap className="h-5 w-5 text-sky-accent" /></StepIcon>
   430→                    <div>
   431→                      <CardTitle className="text-base">Configure &amp; Push</CardTitle>
   432→                      <CardDescription>Review your files and push to GitHub.</CardDescription>
   433→                    </div>
   434→                  </div>
   435→                </CardHeader>
   436→                <CardContent className="space-y-5">
   437→                  {/* File Tree with checkboxes */}
   438→                  <div>
   439→                    <label className="text-sm font-medium mb-2 block">Files Preview <span className="text-muted-foreground font-normal">(click a file to preview)</span></label>
   440→                    <FileTreePreview
   441→                      files={store.files}
   442→                      selectable
   443→                      selectedPaths={store.selectedFilePaths}
   444→                      onToggleFile={store.toggleFilePath}
   445→                      onFileClick={store.setPreviewFile}
   446→                    />
   447→                  </div>
   448→
   449→                  <Separator />
   450→
   451→                  {/* Branch Selector */}
   452→                  <div className="space-y-2">
   453→                    <Label>Branch</Label>
   454→                    <div className="flex gap-2">
   455→                      <Select value={store.branch} onValueChange={store.setBranch}>
   456→                        <SelectTrigger className="flex-1 font-mono text-sm">
   457→                          <SelectValue placeholder={loadingBranches ? 'Loading…' : 'Select branch'} />
   458→                        </SelectTrigger>
   459→                        <SelectContent>
   460→                          {store.branches.map((b) => (
   461→                            <SelectItem key={b} value={b} className="font-mono">{b}</SelectItem>
   462→                          ))}
   463→                        </SelectContent>
   464→                      </Select>
   465→                      <div className="relative flex-1">
   466→                        <Input
   467→                          placeholder="new-branch-name"
   468→                          value={newBranch}
   469→                          onChange={(e) => setNewBranch(e.target.value)}
   470→                          className="font-mono text-sm pr-9"
   471→                          onKeyDown={(e) => e.key === 'Enter' && handleCreateBranch()}
   472→                        />
   473→                        <button
   474→                          type="button"
   475→                          className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-primary transition-colors"
   476→                          onClick={handleCreateBranch}
   477→                          disabled={!newBranch.trim()}
   478→                          aria-label="Create new branch"
   479→                        >
   480→                          <Plus className="h-4 w-4" />
   481→                        </button>
   482→                      </div>
   483→                    </div>
   484→                  </div>
   485→
   486→                  {/* Push Mode */}
   487→                  <div className="space-y-3">
   488→                    <Label>Push Mode</Label>
   489→                    <RadioGroup
   490→                      value={store.mode}
   491→                      onValueChange={(v) => store.setMode(v as PushMode)}
   492→                      className="grid grid-cols-1 sm:grid-cols-2 gap-3"
   493→                    >
   494→                      <label
   495→                        htmlFor="mode-replace"
   496→                        className={cn(
   497→                          'relative flex cursor-pointer flex-col rounded-xl border-2 p-4 transition-all hover:bg-muted/30',
   498→                          store.mode === 'replace' ? 'border-primary bg-primary/5 shadow-sm shadow-primary/5' : 'border-border',
   499→                        )}
   500→                      >
   501→                        <div className="flex items-start gap-2.5">
   502→                          <RadioGroupItem value="replace" id="mode-replace" className="mt-0.5" />
   503→                          <div className="space-y-1">
   504→                            <p className="text-sm font-semibold">Replace Everything</p>
   505→                            <p className="text-xs text-muted-foreground">
   506→                              Completely replace the repository contents. Old files will be removed in a single clean commit.
   507→                            </p>
   508→                          </div>
   509→                        </div>
   510→                        {store.mode === 'replace' && (
   511→                          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute inset-x-0 bottom-0 h-0.5 bg-primary rounded-b-xl" />
   512→                        )}
   513→                      </label>
   514→                      <label
   515→                        htmlFor="mode-smart"
   516→                        className={cn(
   517→                          'relative flex cursor-pointer flex-col rounded-xl border-2 p-4 transition-all hover:bg-muted/30',
   518→                          store.mode === 'smart' ? 'border-sky-accent bg-sky-accent/5 shadow-sm shadow-sky-accent/5' : 'border-border',
   519→                        )}
   520→                      >
   521→                        <div className="flex items-start gap-2.5">
   522→                          <RadioGroupItem value="smart" id="mode-smart" className="mt-0.5" />
   523→                          <div className="space-y-1">
   524→                            <p className="text-sm font-semibold">Smart Update</p>
   525→                            <p className="text-xs text-muted-foreground">Only upload new and changed files. Existing files are preserved.</p>
   526→                          </div>
   527→                        </div>
   528→                        {store.mode === 'smart' && (
   529→                          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute inset-x-0 bottom-0 h-0.5 bg-sky-accent rounded-b-xl" />
   530→                        )}
   531→                      </label>
   532→                    </RadioGroup>
   533→                  </div>
   534→
   535→                  {store.mode === 'replace' && (
   536→                    <div className="flex items-start gap-2 text-xs text-amber-600 dark:text-amber-400 bg-amber-500/10 rounded-lg p-3">
   537→                      <Shield className="h-3.5 w-3.5 mt-0.5 shrink-0" />
   538→                      <p><strong>Destructive action:</strong> This will remove all existing files in the repository that are not in your upload. Make sure you have a backup if needed.</p>
   539→                    </div>
   540→                  )}
   541→
   542→                  {/* Destination Path */}
   543→                  <div className="space-y-2">
   544→                    <Label htmlFor="destination">Destination Subfolder (optional)</Label>
   545→                    <Input
   546→                      id="destination"
   547→                      placeholder="e.g., src/project — leave empty for root"
   548→                      value={store.destination}
   549→                      onChange={(e) => store.setDestination(e.target.value)}
   550→                      className="font-mono text-sm"
   551→                    />
   552→                  </div>
   553→
   554→                  {/* Commit Message */}
   555→                  <div className="space-y-2">
   556→                    <div className="flex items-center justify-between">
   557→                      <Label htmlFor="commit-msg">Commit Message</Label>
   558→                      <Button
   559→                        variant="ghost"
   560→                        size="sm"
   561→                        className="h-6 text-xs gap-1 px-2"
   562→                        onClick={() => {
   563→                          const date = new Date().toISOString().split('T')[0];
   564→                          store.setCommitMessage(`Upload project - ${date}`);
   565→                          toast.success('Suggested a commit message');
   566→                        }}
   567→                      >
   568→                        <Wand2 className="h-3 w-3" />
   569→                        Suggest
   570→                      </Button>
   571→                    </div>
   572→                    <Textarea
   573→                      id="commit-msg"
   574→                      placeholder="Describe your changes..."
   575→                      value={store.commitMessage}
   576→                      onChange={(e) => store.setCommitMessage(e.target.value)}
   577→                      className="min-h-[80px] resize-y text-sm"
   578→                    />
   579→                  </div>
   580→
   581→                  {/* Stats */}
   582→                  <div className="flex flex-wrap gap-2">
   583→                    <Badge variant="secondary" className="text-xs">
   584→                      {selectedFilesForPush.length}/{store.files.length} files selected
   585→                    </Badge>
   586→                    <Badge variant="secondary" className="text-xs">{store.mode === 'replace' ? 'Replace' : 'Smart Update'}</Badge>
   587→                    {store.branch && (
   588→                      <Badge variant="secondary" className="text-xs font-mono">
   589→                        <GitBranch className="mr-1 h-3 w-3" />{store.branch}
   590→                      </Badge>
   591→                    )}
   592→                    {store.destination && (
   593→                      <Badge variant="secondary" className="text-xs font-mono">→ {store.destination}</Badge>
   594→                    )}
   595→                  </div>
   596→
   597→                  {/* Push Button */}
   598→                  <Button
   599→                    size="lg"
   600→                    className="w-full h-12 text-base font-semibold bg-gradient-to-r from-primary to-primary/90 hover:from-primary/90 hover:to-primary shadow-lg shadow-primary/20"
   601→                    onClick={handlePush}
   602→                    disabled={
   603→                      !store.selectedRepo ||
   604→                      selectedFilesForPush.length === 0 ||
   605→                      !store.commitMessage.trim() ||
   606→                      store.stage === 'pushing'
   607→                    }
   608→                  >
   609→                    <Upload className="mr-2 h-5 w-5" />
   610→                    Push {selectedFilesForPush.length} File{selectedFilesForPush.length !== 1 ? 's' : ''} to GitHub
   611→                    <ArrowRight className="ml-2 h-4 w-4" />
   612→                  </Button>
   613→                </CardContent>
   614→              </Card>
   615→            </motion.section>
   616→          )}
   617→        </AnimatePresence>
   618→
   619→        {/* ─── Step 5: Progress ──────────────────────── */}
   620→        <AnimatePresence>
   621→          {store.stage === 'pushing' && (
   622→            <motion.section {...fadeUp} layout>
   623→              <ProgressBar
   624→                stages={store.progressStages}
   625→                currentStage={store.currentProgressStage}
   626→                isComplete={false}
   627→                error={store.pushError}
   628→              />
   629→              {store.pushError && (
   630→                <Button variant="outline" className="w-full mt-4" onClick={() => store.setStage('configure')}>
   631→                  Go Back and Retry
   632→                </Button>
   633→              )}
   634→            </motion.section>
   635→          )}
   636→        </AnimatePresence>
   637→
   638→        {/* ─── Step 6: Success ──────────────────────── */}
   639→        <AnimatePresence>
   640→          {store.stage === 'success' && store.successData && (
   641→            <motion.section {...fadeUp} layout>
   642→              <SuccessScreen
   643→                repoUrl={store.successData.repoUrl}
   644→                commitSha={store.successData.commitSha}
   645→                commitUrl={store.successData.commitUrl}
   646→                commitMessage={store.successData.commitMessage}
   647→                filesUploaded={store.successData.filesUploaded}
   648→                filesChanged={store.successData.filesChanged}
   649→                onPushAnother={() => store.resetPush()}
   650→              />
   651→            </motion.section>
   652→          )}
   653→        </AnimatePresence>
   654→      </div>
   655→
   656→      {/* File Preview Dialog */}
   657→      <FilePreview
   658→        file={store.previewFile}
   659→        open={!!store.previewFile}
   660→        onOpenChange={(open) => { if (!open) store.setPreviewFile(null); }}
   661→      />
   662→
   663→      {/* Create Repo Dialog */}
   664→      <CreateRepoDialog
   665→        open={createRepoOpen}
   666→        onOpenChange={setCreateRepoOpen}
   667→        token={store.token}
   668→        onCreated={handleRepoCreated}
   669→      />
   670→    </div>
   671→  );
   672→}
